export function updateContent() {
    const container = document.getElementById('main-content');
    if (state.activeMainTab === 'dashboard') renderDashboard(container);
    else if (state.activeMainTab === 'database') renderDatabase(container);
}

function getFilteredData() {
    if (!state.startDate || !state.endDate) return state.data;
    return state.data.filter(d => d.colG && d.colG.split('T')[0] >= state.startDate && d.colG.split('T')[0] <= state.endDate);
}

function renderDashboard(container) {
    const filteredData = getFilteredData();
    const totalBaik = filteredData.reduce((sum, item) => sum + (Number(item.colH) || 0), 0);
    const totalRusak = filteredData.reduce((sum, item) => sum + (Number(item.colI) || 0), 0);
    const totalGanti = filteredData.reduce((sum, item) => sum + (Number(item.colJ) || 0), 0);
    const pctGanti = totalBaik > 0 ? ((totalGanti / totalBaik) * 100).toFixed(1) : 0;
    const pctRusak = totalBaik > 0 ? ((totalRusak / totalBaik) * 100).toFixed(1) : 0;

    container.innerHTML = `
        <div class="space-y-6">
            <div class="grid grid-cols-1 md:grid-cols-5 gap-4">
                <div class="bg-white p-4 rounded-xl shadow-sm border">Baik: <h3 class="text-xl font-bold">${totalBaik}</h3></div>
                <div class="bg-white p-4 rounded-xl shadow-sm border">Rusak: <h3 class="text-xl font-bold">${totalRusak}</h3></div>
                <div class="bg-white p-4 rounded-xl shadow-sm border">Ganti: <h3 class="text-xl font-bold">${totalGanti}</h3></div>
                <div class="bg-white p-4 rounded-xl shadow-sm border">% Ganti: <h3 class="text-xl font-bold">${pctGanti}%</h3></div>
                <div class="bg-white p-4 rounded-xl shadow-sm border">% Rusak: <h3 class="text-xl font-bold">${pctRusak}%</h3></div>
            </div>
            <div class="bg-white p-5 rounded-xl shadow-sm border h-72"><canvas id="chartDaily"></canvas></div>
        </div>
    `;
    setTimeout(() => initCharts(filteredData), 50);
}

function initCharts(data) {
    if (state.charts.daily) state.charts.daily.destroy();
    const dailyMap = {};
    data.forEach(d => { if(d.colG) dailyMap[d.colG] = (dailyMap[d.colG] || 0) + (Number(d.colH) || 0); });
    const dates = Object.keys(dailyMap).sort();
    
    const ctx = document.getElementById('chartDaily');
    if (ctx) {
        state.charts.daily = new Chart(ctx, {
            type: 'line',
            data: {
                labels: dates,
                datasets: [{ label: 'Produksi Bagus', data: dates.map(d => dailyMap[d]), borderColor: '#3b82f6', fill: true }]
            },
            options: { responsive: true, maintainAspectRatio: false }
        });
    }
}

function renderDatabase(container) {
    const cols = ['colA', 'colB', 'colC', 'colD', 'colE', 'colF', 'colG'];
    container.innerHTML = `
        <div class="bg-white rounded-xl shadow-sm border overflow-hidden">
            <table class="w-full text-left text-sm">
                <thead class="bg-slate-50 border-b">
                    <tr>${cols.map(c => `<th class="py-3 px-4">${columnLabels[c]}</th>`).join('')}</tr>
                </thead>
                <tbody class="divide-y" id="db-table-body">
                    ${state.data.map((row) => `
                        <tr class="hover:bg-blue-50 cursor-pointer data-row" data-id="${row.colA}">
                            ${cols.map(c => `<td class="py-3 px-4">${row[c] || '-'}</td>`).join('')}
                        </tr>
                    `).join('')}
                </tbody>
            </table>
        </div>
    `;

    document.getElementById('db-table-body').addEventListener('click', (e) => {
        const tr = e.target.closest('.data-row');
        if (tr) openFullRowDetail(tr.dataset.id);
    });
}

export function openFullRowDetail(id) {
    const item = state.data.find(d => d.colA === id);
    if(!item) return;
    const allCols = deptColumns[state.activeSubTab];
    document.getElementById('modal-container').innerHTML = `
        <div id="close-modal" class="fixed inset-0 bg-slate-900/60 flex items-center justify-center z-50 p-4">
            <div class="bg-white p-6 rounded-2xl max-w-2xl w-full max-h-[90vh] overflow-y-auto" onclick="event.stopPropagation()">
                <h3 class="font-bold text-lg mb-4">Rincian Lengkap: ${item.colA}</h3>
                <div class="grid grid-cols-2 gap-3 text-sm">
                    ${allCols.map(k => `<div><strong>${columnLabels[k]}:</strong> ${item[k] || '-'}</div>`).join('')}
                </div>
            </div>
        </div>
    `;
    document.getElementById('close-modal').addEventListener('click', function() {
        this.remove();
    });
}
